# AEGIS - Production Nginx Configuration
# Reverse proxy with SSL termination, rate limiting, and security headers

events {
    worker_connections 1024;
}

http {
    # Security headers
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-Frame-Options "DENY" always;
    add_header X-XSS-Protection "1; mode=block" always;
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains; preload" always;
    add_header Referrer-Policy "strict-origin-when-cross-origin" always;
    add_header Permissions-Policy "camera=(), microphone=(), geolocation=()" always;
    add_header Content-Security-Policy "default-src 'self'; script-src 'self'; style-src 'self' 'unsafe-inline'; img-src 'self' data:; font-src 'self'; object-src 'none'; frame-ancestors 'none'; base-uri 'self'; form-action 'self'" always;

    # Rate limiting zones
    limit_req_zone $binary_remote_addr zone=aegis_api:10m rate=100r/s;
    limit_req_zone $binary_remote_addr zone=aegis_auth:10m rate=20r/s;
    limit_conn_zone $binary_remote_addr zone=aegis_conn:10m max_connections=50;

    # SSL configuration
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384;
    ssl_prefer_server_ciphers on;
    ssl_session_cache shared:SSL:10m;
    ssl_session_timeout 10m;
    ssl_session_tickets off;

    # Logging
    log_format aegis '$remote_addr - $remote_user [$time_local] "$request" '
                     '$status $body_bytes_sent "$http_referer" '
                     '"$http_user_agent" $request_time';
    access_log /var/log/nginx/access.log aegis buffer=32k flush=5s;
    error_log /var/log/nginx/error.log warn;

    # Upstreams
    upstream backend {
        server backend:8000;
        keepalive 32;
    }

    upstream mcp_gateway {
        server mcp-gateway:8443;
        keepalive 16;
    }

    # Main API
    server {
        listen 443 ssl http2;
        server_name api.aegis.security;

        ssl_certificate /etc/nginx/ssl/cert.pem;
        ssl_certificate_key /etc/nginx/ssl/key.pem;

        # Rate limiting
        limit_req zone=aegis_api burst=200 nodelay;
        limit_conn aegis_conn 50;

        # API routes
        location /api/v1/ {
            proxy_pass http://backend;
            proxy_http_version 1.1;
            proxy_set_header Connection "";
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;

            # Timeouts
            proxy_read_timeout 30s;
            proxy_connect_timeout 5s;
            proxy_send_timeout 10s;

            # Body size
            client_max_body_size 10m;
        }

        # MCP Gateway
        location /mcp/ {
            proxy_pass http://mcp_gateway;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
            proxy_read_timeout 60s;
        }

        # Health check (no auth required)
        location /health {
            proxy_pass http://backend/health;
            limit_req zone=aegis_api burst=10 nodelay;
        }

        # Deny all other routes
        location / {
            return 404;
        }
    }

    # HTTP redirect to HTTPS
    server {
        listen 80;
        server_name api.aegis.security;
        return 301 https://$server_name$request_uri;
    }

    # Internal health check (no SSL)
    server {
        listen 8080;
        location /nginx-health {
            access_log off;
            return 200 "healthy\n";
        }
    }
}
